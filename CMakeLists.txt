cmake_minimum_required(VERSION 3.28)
project(FrameGraphLab VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
option(FRAMEGRAPH_BUILD_D3D12 "Build native Windows renderer" ${WIN32})
option(FRAMEGRAPH_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(FRAMEGRAPH_SANITIZERS)
  if(MSVC)
    add_compile_options(/fsanitize=address)
  else()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()
include(CTest)
add_library(framegraph_core src/core/compiler.cpp src/core/validation.cpp src/core/serialization.cpp)
target_include_directories(framegraph_core PUBLIC include)
if(MSVC)
  target_compile_options(framegraph_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(framegraph_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(framegraph_smoke tests/unit/smoke.cpp)
  add_test(NAME baseline_smoke COMMAND framegraph_smoke)
  set_tests_properties(baseline_smoke PROPERTIES TIMEOUT 30)
  add_executable(framegraph_compiler_tests tests/unit/compiler_tests.cpp)
  target_link_libraries(framegraph_compiler_tests PRIVATE framegraph_core)
  add_test(NAME compiler_unit COMMAND framegraph_compiler_tests)
  set_tests_properties(compiler_unit PROPERTIES TIMEOUT 60)
  add_executable(framegraph_property tests/property/graph_sweep.cpp)
  target_link_libraries(framegraph_property PRIVATE framegraph_core)
  add_test(NAME property_smoke COMMAND framegraph_property --cases 256)
  set_tests_properties(property_smoke PROPERTIES TIMEOUT 120)
endif()
