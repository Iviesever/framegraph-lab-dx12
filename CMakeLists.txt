cmake_minimum_required(VERSION 3.28)
project(FrameGraphLab VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
option(FRAMEGRAPH_BUILD_D3D12 "Build native Windows renderer" ${WIN32})
option(FRAMEGRAPH_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(FRAMEGRAPH_SANITIZERS)
  if(MSVC)
    add_compile_options(/fsanitize=address)
  else()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()
include(CTest)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
if(Python3_Interpreter_FOUND)
  execute_process(COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tools/build_contract.py" RESULT_VARIABLE contract_result)
  if(NOT contract_result EQUAL 0)
    message(FATAL_ERROR "Shared MQB/CMake contract rejected")
  endif()
endif()
include(cmake/BuildManifest.cmake)
fgl_add_target(framegraph_core core)
fgl_add_target(framegraph_plan plan)
if(WIN32 AND FRAMEGRAPH_BUILD_D3D12)
  execute_process(COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tools/write_revision.py" RESULT_VARIABLE revision_result)
  if(NOT revision_result EQUAL 0)
    message(FATAL_ERROR "Revision generation failed")
  endif()
  fgl_add_target(FrameGraphLab app)
  fgl_array(runtime_shaders assets shaders)
  foreach(shader IN LISTS runtime_shaders)
    add_custom_command(TARGET FrameGraphLab POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:FrameGraphLab>/shaders"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/${shader}" "$<TARGET_FILE_DIR:FrameGraphLab>/shaders/")
  endforeach()
  if(BUILD_TESTING)
    fgl_add_target(framegraph_shader_tests shader_tests)
    add_test(NAME shader_unit COMMAND framegraph_shader_tests ${CMAKE_CURRENT_SOURCE_DIR})
    set_tests_properties(shader_unit PROPERTIES TIMEOUT 60)
  endif()
endif()
if(BUILD_TESTING)
  fgl_add_target(framegraph_smoke smoke)
  fgl_add_target(framegraph_compiler_tests compiler_tests)
  fgl_add_target(framegraph_property property)
  fgl_add_target(framegraph_planner_tests planner_tests)
  fgl_add_target(framegraph_fuzz fuzz)
  fgl_add_target(framegraph_options_tests options_tests)
  fgl_add_target(framegraph_boundary_tests boundary_tests)
  fgl_add_target(framegraph_expected_fallback_tests expected_fallback_tests)
  add_test(NAME baseline_smoke COMMAND framegraph_smoke)
  add_test(NAME compiler_unit COMMAND framegraph_compiler_tests)
  add_test(NAME property_smoke COMMAND framegraph_property --cases 256)
  add_test(NAME planner_unit COMMAND framegraph_planner_tests)
  add_test(NAME fuzz_smoke COMMAND framegraph_fuzz --iterations 2000)
  add_test(NAME options_unit COMMAND framegraph_options_tests)
  add_test(NAME boundary_unit COMMAND framegraph_boundary_tests)
  add_test(NAME expected_fallback_unit COMMAND framegraph_expected_fallback_tests)
  set_tests_properties(baseline_smoke compiler_unit planner_unit PROPERTIES TIMEOUT 60)
  set_tests_properties(property_smoke fuzz_smoke PROPERTIES TIMEOUT 120)
  if(Python3_Interpreter_FOUND)
    add_test(NAME canonical_json COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/property/canonical_test.py $<TARGET_FILE:framegraph_plan>)
    add_test(NAME build_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/build_contract.py)
    add_test(NAME build_contract_negative COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/property/build_contract_test.py)
    set_tests_properties(canonical_json build_contract PROPERTIES TIMEOUT 90)
  endif()
endif()
fgl_verify_targets()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
install(TARGETS framegraph_core EXPORT FrameGraphLabTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/framegraph DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT FrameGraphLabTargets NAMESPACE FrameGraphLab:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/FrameGraphLab)
configure_package_config_file(cmake/FrameGraphLabConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/FrameGraphLabConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/FrameGraphLab)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/FrameGraphLabConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/FrameGraphLabConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/FrameGraphLabConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/FrameGraphLab)
